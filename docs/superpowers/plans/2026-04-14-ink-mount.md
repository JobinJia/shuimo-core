# InkMount Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a new ink wash mountain rendering engine that produces traditional Chinese painting style mountains with sharp ridges, directional cunfa strokes, ink wash gradients, and atmospheric mist — rendered via Canvas 2D with WebGPU acceleration option.

**Architecture:** Pipeline of pure-data generators (RidgeGenerator → CunFaEngine → InkWashLayer → MistLayer) feeding into swappable render backends. Algorithm layer produces typed data structures; renderer layer draws them. New code lives in `packages/core/src/drawing/ink-mount/`.

**Tech Stack:** TypeScript, Canvas 2D API, WebGPU (optional), SimplexNoise, PerlinNoise, Vector2, seeded PRNG.

// packages/core/src/drawing/ink-mount/RidgeGenerator.ts
import { SimplexNoise } from "../../foundation/noise/SimplexNoise";
import { Vector2 } from "../../foundation/geometry/Vector2";
import type { MountainLayer } from "./types";

export interface RidgeGeneratorInput {
  width: number;
  height: number;
  seed: number;
  depth: number;
  resolution: number;
  peakCount: number;
  sharpness: number;
  subRidgeCount: number;
  noiseOctaves: number;
}

/**
 * Seeded random number generator (simple LCG, no global state)
 */
function seededRandom(seed: number): () => number {
  let s = seed % 2147483647;
  if (s <= 0) s += 2147483646;
  return () => {
    s = (s * 16807) % 2147483647;
    return (s - 1) / 2147483646;
  };
}

/**
 * Generate FBM (Fractional Brownian Motion) value at a point
 */
function fbm(
  noise: SimplexNoise,
  x: number,
  y: number,
  octaves: number,
  persistence: number,
  lacunarity: number,
): number {
  let value = 0;
  let amplitude = 1;
  let frequency = 1;
  let maxAmplitude = 0;

  for (let i = 0; i < octaves; i++) {
    value += noise.noise2D(x * frequency, y * frequency) * amplitude;
    maxAmplitude += amplitude;
    amplitude *= persistence;
    frequency *= lacunarity;
  }

  return value / maxAmplitude;
}

/**
 * Generate a mountain layer with ridge line, sub-ridges, and normals
 */
export function generateRidge(input: RidgeGeneratorInput): MountainLayer {
  const { width, height, seed, depth, resolution, peakCount, sharpness, subRidgeCount, noiseOctaves } = input;
  const noise = new SimplexNoise(seed);
  const rand = seededRandom(seed + 1000);

  const baselineY = height * (0.3 + depth * 0.5);
  const heightAmplitude = height * (0.15 + depth * 0.35);

  // Generate base FBM ridge line
  const ridgeLine: Vector2[] = [];
  for (let i = 0; i < resolution; i++) {
    const t = i / (resolution - 1);
    const x = t * width;
    const noiseVal = fbm(noise, t * 3 + seed * 0.1, depth * 10, noiseOctaves, 0.5, 2.0);
    const y = baselineY - Math.abs(noiseVal) * heightAmplitude;
    ridgeLine.push(new Vector2(x, y));
  }

  // Inject sharp peaks
  if (peakCount > 0) {
    const peakPositions: number[] = [];
    for (let p = 0; p < peakCount; p++) {
      const pos = Math.floor(resolution * (0.15 + rand() * 0.7));
      peakPositions.push(pos);
    }

    for (const peakIdx of peakPositions) {
      const peakHeight = heightAmplitude * (0.3 + rand() * 0.5);
      const falloff = resolution * (0.05 + rand() * 0.1);

      for (let i = 0; i < resolution; i++) {
        const dist = Math.abs(i - peakIdx);
        if (dist < falloff) {
          const factor = Math.pow(1 - dist / falloff, sharpness);
          ridgeLine[i] = new Vector2(ridgeLine[i].x, ridgeLine[i].y - peakHeight * factor);
        }
      }
    }
  }

  // Compute normals (perpendicular to tangent, pointing inward/downward)
  const normals: Vector2[] = [];
  for (let i = 0; i < resolution; i++) {
    const prev = ridgeLine[Math.max(0, i - 1)];
    const next = ridgeLine[Math.min(resolution - 1, i + 1)];
    const tangent = next.sub(prev).normalize();
    // Normal pointing downward into the mountain body
    const normal = new Vector2(tangent.y, -tangent.x);
    // Ensure normal points downward (positive y)
    const corrected = normal.y < 0 ? normal.multiply(-1) : normal;
    normals.push(corrected.normalize());
  }

  // Generate sub-ridges
  const subRidges: Vector2[][] = [];
  for (let s = 0; s < subRidgeCount; s++) {
    const subNoise = new SimplexNoise(seed + 500 + s);
    const decay = 0.4 + s * 0.2;
    const subRidge: Vector2[] = [];

    for (let i = 0; i < resolution; i++) {
      const t = i / (resolution - 1);
      const base = ridgeLine[i];
      const offset = fbm(subNoise, t * 5, s * 3, 4, 0.5, 2.0) * height * 0.05;
      subRidge.push(new Vector2(base.x, base.y + heightAmplitude * decay + offset));
    }
    subRidges.push(subRidge);
  }

  // Compute bounds
  let minX = Infinity,
    minY = Infinity,
    maxX = -Infinity,
    maxY = -Infinity;
  for (const p of ridgeLine) {
    if (p.x < minX) minX = p.x;
    if (p.y < minY) minY = p.y;
    if (p.x > maxX) maxX = p.x;
    if (p.y > maxY) maxY = p.y;
  }

  return {
    depth,
    ridgeLine,
    subRidges,
    normals,
    bounds: { x: minX, y: minY, width: maxX - minX, height: Math.max(maxY, baselineY + height * 0.1) - minY },
  };
}

// packages/core/src/drawing/ink-mount/CunFaEngine.ts
import { Vector2 } from "../../foundation/geometry/Vector2";
import { SimplexNoise } from "../../foundation/noise/SimplexNoise";
import type { CunFaStroke, MountainLayer } from "./types";

export interface CunFaEngineInput {
  layer: MountainLayer;
  seed: number;
  density: number;
  lengthRange: [number, number];
  pressureCurve: number[];
}

function seededRandom(seed: number): () => number {
  let s = seed % 2147483647;
  if (s <= 0) s += 2147483646;
  return () => {
    s = (s * 16807) % 2147483647;
    return (s - 1) / 2147483646;
  };
}

/**
 * Compute curvature at a point on the ridge (higher curvature = denser strokes)
 */
function curvatureAt(points: Vector2[], index: number): number {
  if (index <= 0 || index >= points.length - 1) return 0;
  const prev = points[index - 1];
  const curr = points[index];
  const next = points[index + 1];
  const d1 = curr.sub(prev);
  const d2 = next.sub(curr);
  const cross = Math.abs(d1.x * d2.y - d1.y * d2.x);
  const len = d1.length() * d2.length();
  return len > 0 ? cross / len : 0;
}

/**
 * Interpolate pressure curve at position t (0-1)
 */
function samplePressure(curve: number[], t: number): number {
  if (curve.length === 0) return 1;
  if (curve.length === 1) return curve[0];
  const pos = t * (curve.length - 1);
  const lo = Math.floor(pos);
  const hi = Math.min(lo + 1, curve.length - 1);
  const frac = pos - lo;
  return curve[lo] * (1 - frac) + curve[hi] * frac;
}

/**
 * Generate cunfa (皴法) strokes along the mountain surface normals
 */
export function generateCunFaStrokes(input: CunFaEngineInput): CunFaStroke[] {
  const { layer, seed, density, lengthRange, pressureCurve } = input;
  const { ridgeLine, normals, depth } = layer;
  const rand = seededRandom(seed);
  const noise = new SimplexNoise(seed + 2000);

  const resolution = ridgeLine.length;
  const baseCount = Math.floor(resolution * density * 3);
  const strokes: CunFaStroke[] = [];

  for (let i = 0; i < baseCount; i++) {
    // Pick a point along the ridge
    const ridgeIdx = Math.floor(rand() * (resolution - 2)) + 1;
    const origin = ridgeLine[ridgeIdx];
    const normal = normals[ridgeIdx];

    // Curvature-based density: higher curvature = higher chance of placing stroke
    const curv = curvatureAt(ridgeLine, ridgeIdx);
    const curvThreshold = 0.3 * (1 - density);
    if (rand() > density + curv * 2 && curv < curvThreshold) continue;

    // Stroke length varies by depth (near = longer)
    const [minLen, maxLen] = lengthRange;
    const strokeLen = minLen + rand() * (maxLen - minLen) * (0.5 + depth * 0.5);

    // Direction: along normal with random angular offset ±15°
    const angleOffset = (rand() - 0.5) * Math.PI / 6;
    const dir = normal.rotate(angleOffset);

    // Generate stroke path points
    const pointCount = Math.max(3, Math.floor(strokeLen / 5));
    const path: Vector2[] = [];
    const widths: number[] = [];
    const baseWidth = 1 + rand() * 2;

    for (let j = 0; j < pointCount; j++) {
      const t = j / (pointCount - 1);
      const dist = t * strokeLen;

      // Add slight noise wobble to simulate brush vibration
      const wobble = noise.noise2D(origin.x * 0.01 + j, origin.y * 0.01 + seed) * 3;
      const perpDir = new Vector2(-dir.y, dir.x);

      const point = origin.add(dir.multiply(dist)).add(perpDir.multiply(wobble));
      path.push(point);

      // Pressure: thin → thick → thin
      const pressure = samplePressure(pressureCurve, t);
      widths.push(baseWidth * pressure);
    }

    // Opacity: lighter at peaks (low y), darker at base (high y)
    const normalizedY = (origin.y - layer.bounds.y) / layer.bounds.height;
    const opacity = 0.15 + normalizedY * 0.65 + rand() * 0.15;

    strokes.push({ path, widths, opacity: Math.min(1, opacity) });
  }

  return strokes;
}

// packages/core/src/drawing/ink-mount/InkWashLayer.ts
import { Vector2 } from "../../foundation/geometry/Vector2";
import { SimplexNoise } from "../../foundation/noise/SimplexNoise";
import type { InkFill, MountainLayer } from "./types";

export interface InkWashInput {
  layer: MountainLayer;
  seed: number;
  splashCount: number;
}

function seededRandom(seed: number): () => number {
  let s = seed % 2147483647;
  if (s <= 0) s += 2147483646;
  return () => {
    s = (s * 16807) % 2147483647;
    return (s - 1) / 2147483646;
  };
}

/**
 * Generate a splash (泼墨) region as a closed bezier-like polygon
 */
function generateSplash(cx: number, cy: number, radius: number, seed: number): Vector2[] {
  const noise = new SimplexNoise(seed);
  const points: Vector2[] = [];
  const segments = 12;

  for (let i = 0; i < segments; i++) {
    const angle = (i / segments) * Math.PI * 2;
    const r = radius * (0.5 + 0.5 * Math.abs(noise.noise2D(Math.cos(angle) * 2, Math.sin(angle) * 2)));
    points.push(new Vector2(cx + Math.cos(angle) * r, cy + Math.sin(angle) * r));
  }

  return points;
}

/**
 * Generate ink wash fill data for a mountain layer
 */
export function generateInkFill(input: InkWashInput): InkFill {
  const { layer, seed, splashCount } = input;
  const rand = seededRandom(seed + 3000);
  const { depth, ridgeLine, bounds } = layer;

  // Gradient: mountain top is light, base is dark
  // More layers for near mountains, fewer for far
  const gradientStops = 5;
  const gradient: { stop: number; opacity: number }[] = [];
  const baseOpacity = 0.05 + depth * 0.15;
  const maxOpacity = 0.4 + depth * 0.5;

  for (let i = 0; i < gradientStops; i++) {
    const t = i / (gradientStops - 1);
    const opacity = baseOpacity + (maxOpacity - baseOpacity) * t * t;
    gradient.push({ stop: t, opacity });
  }

  // Splash regions — placed in the lower portion of the mountain
  const splashes: Vector2[][] = [];
  for (let i = 0; i < splashCount; i++) {
    const ridgeIdx = Math.floor(ridgeLine.length * (0.2 + rand() * 0.6));
    const basePoint = ridgeLine[ridgeIdx];
    const splashY = basePoint.y + bounds.height * (0.4 + rand() * 0.4);
    const splashX = basePoint.x + (rand() - 0.5) * bounds.width * 0.15;
    const radius = bounds.height * (0.05 + rand() * 0.1) * (0.5 + depth * 0.5);

    splashes.push(generateSplash(splashX, splashY, radius, seed + 4000 + i));
  }

  return {
    gradient,
    splashes,
    noiseSeed: seed + 5000,
  };
}

// packages/core/src/drawing/ink-mount/MistLayer.ts
import { Vector2 } from "../../foundation/geometry/Vector2";
import { SimplexNoise } from "../../foundation/noise/SimplexNoise";
import type { MistRegion, MountainLayer } from "./types";

export interface MistInput {
  layers: MountainLayer[];
  width: number;
  height: number;
  seed: number;
  opacity: number;
  frequency: number;
  coverage: number;
}

/**
 * Generate mist/fog regions between mountain layers
 */
export function generateMist(input: MistInput): MistRegion[] {
  const { layers, width, height, seed, opacity, frequency, coverage } = input;
  const noise = new SimplexNoise(seed + 6000);
  const regions: MistRegion[] = [];

  // Sort layers by depth (far to near)
  const sorted = [...layers].sort((a, b) => a.depth - b.depth);

  // Generate mist between each adjacent pair of layers
  for (let i = 0; i < sorted.length - 1; i++) {
    const farLayer = sorted[i];
    const nearLayer = sorted[i + 1];

    // Mist band y-range: between the two layers' average ridge heights
    const farAvgY =
      farLayer.ridgeLine.reduce((sum, p) => sum + p.y, 0) / farLayer.ridgeLine.length;
    const nearAvgY =
      nearLayer.ridgeLine.reduce((sum, p) => sum + p.y, 0) / nearLayer.ridgeLine.length;
    const bandTop = Math.min(farAvgY, nearAvgY);
    const bandBottom = Math.max(farAvgY, nearAvgY) + height * 0.1;
    const bandMidY = (bandTop + bandBottom) / 2;
    const bandHeight = bandBottom - bandTop;

    // Number of mist patches in this band
    const patchCount = Math.max(1, Math.floor(3 * coverage));

    for (let p = 0; p < patchCount; p++) {
      const patchSeed = seed + 7000 + i * 100 + p;
      const patchNoise = new SimplexNoise(patchSeed);

      // Horizontally stretched contour
      const contour: Vector2[] = [];
      const segments = 20;
      const cx = width * (0.1 + (p / patchCount) * 0.8);
      const rx = width * (0.15 + coverage * 0.2);
      const ry = bandHeight * (0.2 + coverage * 0.3);

      for (let s = 0; s < segments; s++) {
        const angle = (s / segments) * Math.PI * 2;
        const noiseVal = patchNoise.noise2D(
          Math.cos(angle) * 2,
          Math.sin(angle) * 2,
        );
        const radiusX = rx * (0.7 + 0.3 * Math.abs(noiseVal));
        const radiusY = ry * (0.5 + 0.5 * Math.abs(noiseVal));
        contour.push(
          new Vector2(
            cx + Math.cos(angle) * radiusX,
            bandMidY + Math.sin(angle) * radiusY,
          ),
        );
      }

      // Opacity decreases for bands between far layers
      const bandOpacity = opacity * (0.5 + (1 - farLayer.depth) * 0.5);
      const fadeRadius = Math.max(ry, rx) * 0.4;

      regions.push({
        contour,
        opacity: Math.min(1, bandOpacity),
        fadeRadius,
      });
    }
  }

  return regions;
}

// packages/core/src/drawing/ink-mount/renderer/Canvas2DBackend.ts
import { SimplexNoise } from "../../../foundation/noise/SimplexNoise";
import type { Vector2 } from "../../../foundation/geometry/Vector2";
import type { CunFaStroke, InkFill, MistRegion, MountainLayer } from "../types";
import type { RenderBackend, RenderOutput } from "./types";

export class Canvas2DBackend implements RenderBackend {
  private ctx: CanvasRenderingContext2D;
  private canvas: HTMLCanvasElement | OffscreenCanvas;
  private ownsCanvas: boolean;

  constructor(options: {
    width: number;
    height: number;
    ctx?: CanvasRenderingContext2D;
  }) {
    if (options.ctx) {
      this.ctx = options.ctx;
      this.canvas = options.ctx.canvas;
      this.ownsCanvas = false;
    } else {
      if (typeof OffscreenCanvas !== "undefined") {
        this.canvas = new OffscreenCanvas(options.width, options.height);
      } else {
        this.canvas = document.createElement("canvas");
        this.canvas.width = options.width;
        this.canvas.height = options.height;
      }
      this.ctx = this.canvas.getContext("2d")!;
      this.ownsCanvas = true;
    }
  }

  clear(): void {
    const { width, height } = this.canvas;
    this.ctx.clearRect(0, 0, width, height);
  }

  drawMountainFill(layer: MountainLayer, ink: InkFill): void {
    const ctx = this.ctx;
    const { ridgeLine, bounds, depth } = layer;
    const canvasHeight = this.canvas.height;

    // Build mountain polygon path (ridge line + bottom edge)
    ctx.save();
    ctx.beginPath();
    ctx.moveTo(ridgeLine[0].x, ridgeLine[0].y);
    for (let i = 1; i < ridgeLine.length; i++) {
      ctx.lineTo(ridgeLine[i].x, ridgeLine[i].y);
    }
    // Close along the bottom
    ctx.lineTo(ridgeLine[ridgeLine.length - 1].x, canvasHeight);
    ctx.lineTo(ridgeLine[0].x, canvasHeight);
    ctx.closePath();
    ctx.clip();

    // Draw gradient fill
    const gradY0 = bounds.y;
    const gradY1 = bounds.y + bounds.height;
    const gradient = ctx.createLinearGradient(0, gradY0, 0, gradY1);
    for (const stop of ink.gradient) {
      const gray = Math.floor(20 - depth * 10);
      gradient.addColorStop(stop.stop, `rgba(${gray},${gray},${gray + 5},${stop.opacity})`);
    }
    ctx.fillStyle = gradient;
    ctx.fillRect(bounds.x, bounds.y, bounds.width, canvasHeight - bounds.y);

    // Draw ink noise texture
    const noise = new SimplexNoise(ink.noiseSeed);
    const step = 8;
    for (let y = bounds.y; y < canvasHeight; y += step) {
      for (let x = bounds.x; x < bounds.x + bounds.width; x += step) {
        const n = noise.noise2D(x * 0.008, y * 0.008);
        if (n > 0.1) {
          const alpha = n * 0.15 * (0.5 + depth * 0.5);
          ctx.fillStyle = `rgba(5,5,10,${alpha})`;
          ctx.fillRect(x, y, step, step);
        }
      }
    }

    // Draw splash regions
    for (const splash of ink.splashes) {
      ctx.beginPath();
      ctx.moveTo(splash[0].x, splash[0].y);
      for (let i = 1; i < splash.length; i++) {
        const prev = splash[i - 1];
        const curr = splash[i];
        const next = splash[(i + 1) % splash.length];
        const cpx = curr.x;
        const cpy = curr.y;
        const ex = (curr.x + next.x) / 2;
        const ey = (curr.y + next.y) / 2;
        ctx.quadraticCurveTo(cpx, cpy, ex, ey);
      }
      ctx.closePath();
      ctx.fillStyle = `rgba(5,5,10,${0.3 + depth * 0.4})`;
      ctx.fill();
    }

    ctx.restore();
  }

  drawCunFaStrokes(strokes: CunFaStroke[]): void {
    const ctx = this.ctx;

    for (const stroke of strokes) {
      const { path, widths, opacity } = stroke;
      if (path.length < 2) continue;

      ctx.save();
      ctx.lineCap = "round";
      ctx.lineJoin = "round";
      ctx.strokeStyle = `rgba(10,10,15,${opacity})`;

      // Draw each segment with varying width
      for (let i = 0; i < path.length - 1; i++) {
        ctx.beginPath();
        ctx.lineWidth = widths[i];
        ctx.moveTo(path[i].x, path[i].y);
        ctx.lineTo(path[i + 1].x, path[i + 1].y);
        ctx.stroke();
      }

      ctx.restore();
    }
  }

  drawMist(regions: MistRegion[]): void {
    const ctx = this.ctx;

    for (const region of regions) {
      const { contour, opacity, fadeRadius } = region;
      if (contour.length < 3) continue;

      // Calculate centroid
      let cx = 0,
        cy = 0;
      for (const p of contour) {
        cx += p.x;
        cy += p.y;
      }
      cx /= contour.length;
      cy /= contour.length;

      ctx.save();

      // Build path
      ctx.beginPath();
      ctx.moveTo(contour[0].x, contour[0].y);
      for (let i = 1; i < contour.length; i++) {
        ctx.lineTo(contour[i].x, contour[i].y);
      }
      ctx.closePath();

      // Radial gradient for soft edges
      const maxRadius = fadeRadius * 2;
      const gradient = ctx.createRadialGradient(cx, cy, 0, cx, cy, maxRadius);
      gradient.addColorStop(0, `rgba(255,255,255,${opacity})`);
      gradient.addColorStop(0.6, `rgba(255,255,255,${opacity * 0.5})`);
      gradient.addColorStop(1, `rgba(255,255,255,0)`);

      ctx.fillStyle = gradient;
      ctx.fill();

      ctx.restore();
    }
  }

  drawRidgeLine(points: Vector2[], opacity: number, lineWidth: number): void {
    const ctx = this.ctx;
    if (points.length < 2) return;

    ctx.save();
    ctx.strokeStyle = `rgba(15,15,20,${opacity})`;
    ctx.lineWidth = lineWidth;
    ctx.lineCap = "round";
    ctx.lineJoin = "round";

    ctx.beginPath();
    ctx.moveTo(points[0].x, points[0].y);
    for (let i = 1; i < points.length; i++) {
      ctx.lineTo(points[i].x, points[i].y);
    }
    ctx.stroke();

    ctx.restore();
  }

  toOutput(): RenderOutput {
    return { type: "canvas", canvas: this.canvas };
  }
}

// packages/core/src/drawing/ink-mount/InkMount.ts
import { generateRidge } from "./RidgeGenerator";
import { generateCunFaStrokes } from "./CunFaEngine";
import { generateInkFill } from "./InkWashLayer";
import { generateMist } from "./MistLayer";
import { Canvas2DBackend } from "./renderer/Canvas2DBackend";
import type {
  InkMountOptions,
  InkMountLayerOptions,
  InkMountScene,
  QualityPreset,
  RidgeOptions,
  CunFaOptions,
  MistOptions,
} from "./types";
import type { RenderBackend, RenderOutput } from "./renderer/types";

const QUALITY_PRESETS: Record<QualityPreset, { density: number; octaves: number; splashCount: number }> = {
  draft: { density: 0.2, octaves: 4, splashCount: 0 },
  normal: { density: 0.5, octaves: 6, splashCount: 3 },
  high: { density: 0.85, octaves: 8, splashCount: 5 },
};

const DEFAULT_RIDGE: RidgeOptions = {
  peakCount: 2,
  sharpness: 3,
  subRidgeCount: 2,
  noiseOctaves: 6,
};

const DEFAULT_CUNFA: CunFaOptions = {
  density: 0.5,
  lengthRange: [15, 60],
  pressureCurve: [0.3, 1.0, 0.3],
};

const DEFAULT_MIST: MistOptions = {
  opacity: 0.8,
  frequency: 0.005,
  coverage: 0.6,
};

function autoLayerCount(height: number): number {
  return Math.min(10, Math.max(2, Math.floor(height / 120)));
}

export class InkMount {
  /**
   * Generate scene data (pure data, no rendering)
   */
  static generateScene(options: InkMountOptions): InkMountScene {
    const { width, height, seed } = options;
    const quality = QUALITY_PRESETS[options.quality ?? "normal"];
    const ridgeOpts = { ...DEFAULT_RIDGE, ...options.ridge };
    const cunfaOpts = { ...DEFAULT_CUNFA, ...options.cunfa, density: options.cunfa?.density ?? quality.density };
    const mistOpts = { ...DEFAULT_MIST, ...options.mist };

    ridgeOpts.noiseOctaves = options.ridge?.noiseOctaves ?? quality.octaves;

    const layerCount = options.layers ?? autoLayerCount(height);

    // Generate layers from far (0) to near (1)
    const layers = [];
    const allStrokes = [];
    const allFills = [];

    for (let i = 0; i < layerCount; i++) {
      const depth = layerCount === 1 ? 0.5 : i / (layerCount - 1);
      const layer = generateRidge({
        width,
        height,
        seed: seed + i * 137,
        depth,
        resolution: 200,
        peakCount: ridgeOpts.peakCount,
        sharpness: ridgeOpts.sharpness,
        subRidgeCount: ridgeOpts.subRidgeCount,
        noiseOctaves: ridgeOpts.noiseOctaves,
      });

      const strokes = generateCunFaStrokes({
        layer,
        seed: seed + i * 251,
        density: cunfaOpts.density,
        lengthRange: cunfaOpts.lengthRange,
        pressureCurve: cunfaOpts.pressureCurve,
      });

      const fill = generateInkFill({
        layer,
        seed: seed + i * 389,
        splashCount: quality.splashCount,
      });

      layers.push(layer);
      allStrokes.push(strokes);
      allFills.push(fill);

      if (options.onLayer) {
        options.onLayer(layer, i);
      }
    }

    const mists = generateMist({
      layers,
      width,
      height,
      seed: seed + 9999,
      opacity: mistOpts.opacity,
      frequency: mistOpts.frequency,
      coverage: mistOpts.coverage,
    });

    return { layers, strokes: allStrokes, fills: allFills, mists, width, height };
  }

  /**
   * Generate a single layer scene (for composability)
   */
  static generateLayerScene(options: InkMountLayerOptions): InkMountScene {
    const { width, height, seed, depth } = options;
    const quality = QUALITY_PRESETS[options.quality ?? "normal"];
    const ridgeOpts = { ...DEFAULT_RIDGE, ...options.ridge };
    const cunfaOpts = { ...DEFAULT_CUNFA, ...options.cunfa, density: options.cunfa?.density ?? quality.density };

    const layer = generateRidge({
      width,
      height,
      seed,
      depth,
      resolution: 200,
      peakCount: ridgeOpts.peakCount,
      sharpness: ridgeOpts.sharpness,
      subRidgeCount: ridgeOpts.subRidgeCount,
      noiseOctaves: ridgeOpts.noiseOctaves,
    });

    const strokes = generateCunFaStrokes({
      layer,
      seed: seed + 251,
      density: cunfaOpts.density,
      lengthRange: cunfaOpts.lengthRange,
      pressureCurve: cunfaOpts.pressureCurve,
    });

    const fill = generateInkFill({
      layer,
      seed: seed + 389,
      splashCount: quality.splashCount,
    });

    return { layers: [layer], strokes: [strokes], fills: [fill], mists: [], width, height };
  }

  /**
   * Generate and render a complete scene to canvas
   */
  static generate(options: InkMountOptions): RenderOutput {
    const scene = InkMount.generateScene(options);
    const backend = InkMount.createBackend(options);
    InkMount.renderScene(scene, backend);
    return backend.toOutput();
  }

  /**
   * Generate and render a single layer to canvas
   */
  static generateLayer(options: InkMountLayerOptions): RenderOutput {
    const scene = InkMount.generateLayerScene(options);
    const backend = InkMount.createBackend(options);
    InkMount.renderScene(scene, backend);
    return backend.toOutput();
  }

  /**
   * Render a pre-generated scene to a backend
   */
  static renderScene(scene: InkMountScene, backend: RenderBackend): void {
    backend.clear();

    // Render back-to-front
    for (let i = 0; i < scene.layers.length; i++) {
      const layer = scene.layers[i];

      // Draw mountain fill
      backend.drawMountainFill(layer, scene.fills[i]);

      // Draw cunfa strokes
      backend.drawCunFaStrokes(scene.strokes[i]);

      // Draw ridge outline
      const ridgeOpacity = 0.2 + layer.depth * 0.5;
      const ridgeWidth = 0.5 + layer.depth * 1.0;
      backend.drawRidgeLine(layer.ridgeLine, ridgeOpacity, ridgeWidth);

      // Draw sub-ridges
      for (const sub of layer.subRidges) {
        backend.drawRidgeLine(sub, ridgeOpacity * 0.5, ridgeWidth * 0.6);
      }

      // Draw mist after each layer (except last)
      if (i < scene.layers.length - 1) {
        const layerMists = scene.mists.filter((_, mi) => mi === i);
        if (layerMists.length > 0) {
          backend.drawMist(layerMists);
        }
      }
    }

    // Draw remaining mists on top
    if (scene.mists.length > scene.layers.length - 1) {
      backend.drawMist(scene.mists.slice(scene.layers.length - 1));
    }
  }

  private static createBackend(options: { width: number; height: number; ctx?: CanvasRenderingContext2D; backend?: string }): RenderBackend {
    // TODO: WebGPU detection and backend selection in future task
    return new Canvas2DBackend({
      width: options.width,
      height: options.height,
      ctx: options.ctx,
    });
  }
}

<!-- playground/src/demos/InkMountDemo.vue -->
<script setup lang="ts">
import { onMounted, ref } from "vue";
import { InkMount } from "@jobinjia/shuimo-core";

const canvasRef = ref<HTMLCanvasElement>();
const seed = ref(42);
const layers = ref(5);
const quality = ref<"draft" | "normal" | "high">("high");

function render() {
  const canvas = canvasRef.value;
  if (!canvas) return;

  const ctx = canvas.getContext("2d");
  if (!ctx) return;

  canvas.width = 1200;
  canvas.height = 800;

  ctx.fillStyle = "#faf8f5";
  ctx.fillRect(0, 0, canvas.width, canvas.height);

  InkMount.generate({
    width: canvas.width,
    height: canvas.height,
    seed: seed.value,
    layers: layers.value,
    quality: quality.value,
    ctx,
  });
}

onMounted(render);

function regenerate() {
  seed.value = Math.floor(Math.random() * 100000);
  render();
}
</script>

<template>
  <div style="padding: 20px">
    <h2>InkMount — 写意水墨山</h2>
    <div style="margin-bottom: 12px; display: flex; gap: 12px; align-items: center">
      <label>
        Layers:
        <input v-model.number="layers" type="range" min="2" max="10" step="1" @input="render" />
        {{ layers }}
      </label>
      <label>
        Quality:
        <select v-model="quality" @change="render">
          <option value="draft">Draft</option>
          <option value="normal">Normal</option>
          <option value="high">High</option>
        </select>
      </label>
      <label>
        Seed:
        <input v-model.number="seed" type="number" style="width: 80px" @change="render" />
      </label>
      <button @click="regenerate">Regenerate</button>
    </div>
    <canvas
      ref="canvasRef"
      style="border: 1px solid #ddd; max-width: 100%; background: #faf8f5"
    />
  </div>
</template>
